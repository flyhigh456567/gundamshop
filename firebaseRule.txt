Firebase Firestore Rules
========================

貼到 Firebase Console > Firestore Database > Rules，然後發布。
管理員角色存在 users/{UID}.role。只有你在 Firebase Console 手動改成 admin 的會員才可管理商品；前端不能自行升級角色。
商品資料儲存在 Firestore 的 products 集合。商品圖片改由 GitHub Pages 提供，不使用 Firebase Storage；因此不需要另外設定或發布 Firebase Storage Rules。

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return signedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    match /users/{userId} {
      allow create: if signedIn()
        && request.auth.uid == userId
        && request.resource.data.keys().hasOnly([
          'displayName', 'email', 'phone', 'address', 'role', 'createdAt', 'updatedAt'
        ])
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.role == 'user'
        && request.resource.data.displayName == request.resource.data.displayName.lower()
        && request.resource.data.displayName.matches('^[^/]{1,60}$')
        && getAfter(/databases/$(database)/documents/memberNames/$(request.resource.data.displayName)).data.uid == userId;

      allow read: if signedIn() && (request.auth.uid == userId || isAdmin());

      allow update: if signedIn()
        && request.auth.uid == userId
        && request.resource.data.keys().hasOnly([
          'displayName', 'email', 'phone', 'address', 'role', 'createdAt', 'updatedAt'
        ])
        && request.resource.data.email == resource.data.email
        && request.resource.data.get('role', 'user') == resource.data.get('role', 'user')
        && request.resource.data.get('createdAt', resource.data.get('createdAt', null))
          == resource.data.get('createdAt', null)
        && request.resource.data.displayName == request.resource.data.displayName.lower()
        && request.resource.data.displayName.matches('^[^/]{1,60}$')
        && getAfter(/databases/$(database)/documents/memberNames/$(request.resource.data.displayName)).data.uid == userId;

      // Role changes are deliberately limited to trusted Firebase Console/Admin SDK operations.
      allow delete: if isAdmin();
    }

    match /memberNames/{nameKey} {
      // Names are case-normalized by the client and stored lowercase. A signed-in user
      // can check one name but cannot list the registry.
      allow get: if signedIn();
      allow list: if false;
      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(['uid'])
        && request.resource.data.uid == request.auth.uid
        && getAfter(/databases/$(database)/documents/users/$(request.auth.uid)).data.displayName == nameKey;
      allow update: if false;
      allow delete: if signedIn()
        && resource.data.uid == request.auth.uid
        && getAfter(/databases/$(database)/documents/users/$(request.auth.uid)).data.displayName != nameKey;
    }

    match /products/{productId} {
      allow read: if isAdmin() || resource.data.isPublished == true;
      allow create, update: if isAdmin()
        && request.resource.data.keys().hasAll(['image_url'])
        && (request.resource.data.image_url == null
          || (request.resource.data.image_url is string
            && request.resource.data.image_url.matches('^[^/]{1,200}$')));
      allow delete: if isAdmin();
    }

    match /orders/{orderId} {
      allow create: if signedIn()
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.keys().hasOnly([
          'userId', 'orderNo', 'customer', 'items', 'subtotal', 'shipping',
          'total', 'paymentMethod', 'status', 'createdAt'
        ])
        && request.resource.data.status == 'pending_payment'
        && request.resource.data.paymentMethod == 'unintegrated_test';

      allow read: if signedIn()
        && (resource.data.userId == request.auth.uid || isAdmin());
      allow update, delete: if isAdmin();
    }
  }
}

如何設定管理員
==============

1. 先在網站註冊並登入該會員一次。註冊會建立 Authentication 帳號及 Firestore users/{UID} 文件，欄位 role 初始值為 user。
2. Firebase Console > Firestore Database > users > 該會員 UID 文件，將 role 字串從 user 改成 admin。
3. 重新整理網站或重新登入，管理控制台就會顯示。商品新增、編輯及刪除都由 Firestore Rules 檢查 admin 權限。
4. 不要讓一般會員具備 Firebase Console 權限；會員只能讀取自己的角色，Rules 會拒絕自行修改 role。
